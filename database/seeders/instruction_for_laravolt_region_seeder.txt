<?php

namespace Database\Seeders;

use Illuminate\Database\Seeder;
use Illuminate\Support\Facades\DB;

class RegionFullSeeder extends Seeder
{
    public function run(): void
    {
        // Hapus data lama
        DB::table('regions')->truncate();

        $this->command->info('Seeding provinces...');
        $provinces = DB::table('laravolt_indonesia_provinces')->get();
        
        foreach ($provinces as $province) {
            DB::table('regions')->insert([
                'region_type' => 'province',
                'code'        => $province->code,
                'name'        => $province->name,
                'parent_id'   => null,
                'created_at'  => now(),
                'updated_at'  => now(),
            ]);
        }

        $this->command->info('Seeding regencies...');
        $regencies = DB::table('laravolt_indonesia_regencies')->get();

        foreach ($regencies as $regency) {
            // Cari province_id di tabel regions berdasarkan code
            $province = DB::table('regions')
                ->where('region_type', 'province')
                ->where('code', $regency->province_code)
                ->first();

            if (!$province) continue;

            DB::table('regions')->insert([
                'region_type' => 'regency',
                'code'        => $regency->code,
                'name'        => $regency->name,
                'parent_id'   => $province->id,
                'created_at'  => now(),
                'updated_at'  => now(),
            ]);
        }

        $this->command->info('Seeding districts...');
        $districts = DB::table('laravolt_indonesia_districts')->get();

        foreach ($districts as $district) {
            $regency = DB::table('regions')
                ->where('region_type', 'regency')
                ->where('code', $district->regency_code)
                ->first();

            if (!$regency) continue;

            DB::table('regions')->insert([
                'region_type' => 'district',
                'code'        => $district->code,
                'name'        => $district->name,
                'parent_id'   => $regency->id,
                'created_at'  => now(),
                'updated_at'  => now(),
            ]);
        }

        $this->command->info('✅ Selesai! Data wilayah Indonesia berhasil diimport.');
    }
}

// run seeder
php artisan db:seed --class=RegionFullSeeder
